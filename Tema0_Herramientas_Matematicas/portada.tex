\begin{tikzpicture}[remember picture, overlay]
  \coordinate (O) at (current page.south west);

  % ----------------------------------------------------------------
  % Motivo gráfico: curvas de nivel de f = exp(-rho^2/s^2) (equiespaciadas
  % en f) y campo gradiente (flechas hacia el máximo, longitud ~ |grad f|).
  % ----------------------------------------------------------------
  \begin{scope}
    \clip (O) rectangle ($(O)+(21cm,24.6cm)$);
    % curvas de nivel
    \foreach \k in {1,...,14}{
      \pgfmathsetmacro{\fk}{\k/15}
      \pgfmathsetmacro{\rr}{5.2*sqrt(-ln(\fk))}
      \pgfmathsetmacro{\pc}{100-6.2*\k}
      \pgfmathsetmacro{\lw}{0.25+0.045*\k}
      \draw[azulprof!\pc!azulclaro, line width=\lw pt]
        ($(O)+(14.2cm,17.2cm)$) circle (\rr cm);
    }
    % flechas del gradiente a lo largo de 24 rayos
    \foreach \ang in {0,15,...,345}{
      \foreach \rho in {1.6,2.9,4.3,6.1,8.3}{
        \pgfmathsetmacro{\g}{(\rho/5.2)*exp(-(\rho/5.2)*(\rho/5.2))}
        \pgfmathsetmacro{\len}{3.2*\g}
        \pgfmathsetmacro{\pc}{30+250*\g}
        \draw[-{Stealth[length=3.2pt,width=3pt]}, line width=0.5pt,
              azulacero!\pc!azulclaro]
          ($(O)+(14.2cm,17.2cm)+(\ang:\rho cm)$) -- ++(\ang+180:\len cm);
      }
    }
    \fill[azulprof] ($(O)+(14.2cm,17.2cm)$) circle (2.2pt);
  \end{scope}

  % ----------------------------------------------------------------
  % Cabecera: logotipos y filete
  % ----------------------------------------------------------------
  \node[anchor=north west, inner sep=0] at ($(O)+(2.2cm,27.9cm)$)
    {\includegraphics[height=1.42cm]{logo_usal.png}};
  \node[anchor=north east, inner sep=0] at ($(O)+(18.8cm,27.9cm)$)
    {\includegraphics[height=1.42cm]{logo_facultad.png}};
  \draw[grispizarra!60, line width=0.4pt]
    ($(O)+(2.2cm,26.0cm)$) -- ($(O)+(18.8cm,26.0cm)$);
  \node[anchor=north west, inner sep=0, text=grispizarra, font=\scshape\small]
    at ($(O)+(2.2cm,25.75cm)$)
    {\textls[80]{\Universidad\ \textbar\ \Facultad}};
  \node[anchor=north east, inner sep=0, text=grispizarra, font=\scshape\small]
    at ($(O)+(18.8cm,25.75cm)$)
    {\textls[80]{\Titulacion}};

  % ----------------------------------------------------------------
  % Bloque del título (sobre un velo blanco para que el motivo no tape el texto)

    ($(O)+(0,12.3cm)$) rectangle ($(O)+(21cm,13.4cm)$);

  % Perfil de f a lo largo de un diámetro, como filete ondulado

    \draw[azulacero, line width=0.9pt, domain=2.2:18.8, samples=300, smooth]

  \end{scope}

  \node[anchor=north west, inner sep=0, text width=16.6cm, align=left]
    at ($(O)+(2.2cm,12.0cm)$)
    {{\color{azulacero}\scshape\large\textls[120]{\Asignatura\ \textbar\ \NumTema}}\\[12pt]

     {\color{grispizarra}\itshape\large
      $\displaystyle \int_{\vc{a}}^{\vc{b}}\nab f\cdot\dl \;=\; f(\vc{b})-f(\vc{a})$}};
