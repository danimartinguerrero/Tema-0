% Figura: verificación del teorema del gradiente por dos caminos
\begin{figure}[ht]
\centering
\tdplotsetmaincoords{62}{112}
\begin{tikzpicture}[tdplot_main_coords, scale=1.9]
  \draw[eje] (0,0,0) -- (2.8,0,0) node[below left] {\small $x$};
  \draw[eje] (0,0,0) -- (0,2.8,0) node[right] {\small $y$};
  \draw[eje] (0,0,0) -- (0,0,2.8) node[above] {\small $z$};
  % aristas del cubo como referencia
  \draw[grisclaro, line width=0.4pt] (2,0,0) -- (2,2,0) -- (0,2,0) (2,2,0) -- (2,2,2) (2,0,0) -- (2,0,2) -- (2,2,2) (0,2,0) -- (0,2,2) -- (2,2,2) (0,0,2) -- (2,0,2) (0,0,2) -- (0,2,2);
  % camino 1: segmento recto
  \draw[line width=1.1pt, acento] (0,0,0) -- (2,2,2);
  \draw[-{Stealth[length=5pt]}, line width=1.1pt, acento] (0.95,0.95,0.95) -- (1.05,1.05,1.05);
  % camino 2: poligonal por los ejes
  \draw[line width=1.1pt, azulprof] (0,0,0) -- (2,0,0) -- (2,2,0) -- (2,2,2);
  \draw[-{Stealth[length=5pt]}, line width=1.1pt, azulprof] (0.95,0,0) -- (1.05,0,0);
  \draw[-{Stealth[length=5pt]}, line width=1.1pt, azulprof] (2,0.95,0) -- (2,1.05,0);
  \draw[-{Stealth[length=5pt]}, line width=1.1pt, azulprof] (2,2,0.95) -- (2,2,1.05);
  % extremos, discretos
  \fill[azulprof] (0,0,0) circle (1.1pt);
  \fill[azulprof] (2,2,2) circle (1.1pt);
  \node[font=\small, anchor=east] at (-0.05,0,0) {$\vc{a}=(0,0,0)$};
  \node[font=\small, anchor=south west] at (2.0,2.0,2.08) {$\vc{b}=(2,2,2)$};
  \node[font=\small, text=acento, anchor=south east] at (0.9,0.95,1.35) {$\gamma_1$};
  \node[font=\small, text=azulprof, anchor=north east] at (1,0,-0.05) {$C_x$};
  \node[font=\small, text=azulprof, anchor=west] at (2.15,1.0,-0.1) {$C_y$};
  \node[font=\small, text=azulprof, anchor=west] at (2.1,2.02,0.9) {$C_z$};
\end{tikzpicture}
\caption{Verificación del teorema del gradiente para $f=xy^{2}$ entre
$\vc{a}=(0,0,0)$ y $\vc{b}=(2,2,2)$. En ocre, el segmento $\gamma_1$; en azul, la
ruta escalonada $\gamma_2=C_x\uplus C_y\uplus C_z$. En $C_x$ el gradiente $\nab f=(y^{2},2xy,0)$ es nulo y en $C_z$ es
perpendicular al camino, de modo que el incremento $f=0\to8$ lo aporta solo
$C_y$. Ambos caminos dan $\int\nab f\cdot\dl=8$.}
\label{fig:sem-trayectorias}
\end{figure}
